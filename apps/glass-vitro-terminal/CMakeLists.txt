cmake_minimum_required(VERSION 3.20)
project(GLASS_VITROMATON LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(unison_core STATIC core/unison.cpp core/pixels.cpp)
target_include_directories(unison_core PUBLIC core)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(unison_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NOT ANDROID)
 add_executable(glass-terminal cli/terminal.cpp)
 target_link_libraries(glass-terminal PRIVATE unison_core)
 add_executable(glass-tests tests/tests.cpp)
 target_link_libraries(glass-tests PRIVATE unison_core)
 enable_testing()
 add_test(NAME deterministic_oracles COMMAND glass-tests)
endif()
